let sanitize s = String.map (fun c -> match c with 'a'..'z' | 'A'..'Z' | '0'..'9' | '-' -> c | _ -> '_') s

let write dir name ext text =
  let path = Filename.concat dir (sanitize name ^ ext) in
  Out_channel.with_open_bin path (fun c -> output_string c text)

let () =
  let dir = Sys.argv.(1) in
  List.iter (fun sub -> try Sys.mkdir (Filename.concat dir sub) 0o755 with _ -> ()) [ "wgsl"; "wgsl_fast"; "js_region"; "js_region_fast"; "wgsl_error" ];
  let all_compiled =
    List.map (fun (n, s) -> ("f_" ^ n, Test_program.compile s)) fast_cases
    @ List.map (fun (n, p) -> ("c4096_" ^ n, p)) (Test_program.carried_fixtures 4096)
  in
  List.iter
    (fun (name, program) ->
      (match Flow_web.Wgsl.kernel_fast program with
       | Ok w -> write (Filename.concat dir "wgsl_fast") name ".wgsl" (w.entry ^ "\n" ^ w.source)
       | Error d -> write (Filename.concat dir "wgsl_fast") name ".wgsl" ("ERROR " ^ d.code ^ " " ^ d.message));
      write (Filename.concat dir "js_region_fast") name ".js" ((try Flow_web.Js.region ~fast:true program with Invalid_argument m -> "EXN Invalid_argument " ^ m));
      write (Filename.concat dir "js_region") name ".js" ((try Flow_web.Js.region program with Invalid_argument m -> "EXN Invalid_argument " ^ m)))
    all_compiled;
  List.iter
    (fun (name, source) ->
      let program = Test_program.compile source in
      (match Flow_web.Wgsl.kernel program with
       | Ok w -> write (Filename.concat dir "wgsl") name ".wgsl" (w.entry ^ "\n" ^ w.source)
       | Error d -> write (Filename.concat dir "wgsl") name ".wgsl" ("ERROR " ^ d.code ^ " " ^ d.message));
      write (Filename.concat dir "js_region") name ".js" ((try Flow_web.Js.region program with Invalid_argument m -> "EXN Invalid_argument " ^ m));
      write (Filename.concat dir "js_region_fast") name ".js" ((try Flow_web.Js.region ~fast:true program with Invalid_argument m -> "EXN Invalid_argument " ^ m)))
    (List.map (fun (n, s) -> (n, s)) [])
  ;
  List.iter
    (fun (name, program) ->
      write (Filename.concat dir "js_region") name ".js" ((try Flow_web.Js.region program with Invalid_argument m -> "EXN Invalid_argument " ^ m));
      write (Filename.concat dir "js_region_fast") name ".js" ((try Flow_web.Js.region ~fast:true program with Invalid_argument m -> "EXN Invalid_argument " ^ m));
      match Flow_web.Wgsl.kernel program with
